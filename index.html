        // Pool of fallback accounts

        const fallbackAccounts = Array.from(new Set(

            transactions.flatMap(t => [t.dr, t.cr, ...t.wrong])

        ));

 

        // State variables

        let currentIndex = 0;

        let score = 0;

        let currentStage = 'Dr'; // 'Dr' or 'Cr'

        let hasFailedCurrent = false; // Endless trial check

        let correctAnswers = [];

        let wrongAnswers = [];

        let currentChoices = [];

 

        // Audio Management

        const bgmAudio = document.getElementById('bgm-audio');

        const dingAudio = document.getElementById('ding-audio');

        const incorrectAudio = document.getElementById('incorrect-audio');

        const bgmToggle = document.getElementById('bgm-toggle');

        let bgmEnabled = true;

 

        bgmAudio.volume = 0.25;

 

        function toggleBGM() {

            bgmEnabled = !bgmEnabled;

            if (bgmEnabled) {

                bgmToggle.textContent = "🔊 BGM";

                if (document.getElementById('game-screen').classList.contains('active') ||

                    document.getElementById('instruction-screen').classList.contains('active')) {

                    bgmAudio.play().catch(() => {});

                }

            } else {

                bgmToggle.textContent = "🔇 BGM";

                bgmAudio.pause();

            }

        }

 

        bgmToggle.addEventListener('click', toggleBGM);

 

        // Web Audio Synthesizer for gunshot

        const audioCtx = new (window.AudioContext || window.webkitAudioContext)();

 

        function playGunshot() {

            if (audioCtx.state === 'suspended') audioCtx.resume();

            const bufferSize = audioCtx.sampleRate * 0.12;

            const buffer = audioCtx.createBuffer(1, bufferSize, audioCtx.sampleRate);

            const data = buffer.getChannelData(0);

            for (let i = 0; i < bufferSize; i++) data[i] = Math.random() * 2 - 1;

 

            const noise = audioCtx.createBufferSource();

            noise.buffer = buffer;

 

            const filter = audioCtx.createBiquadFilter();

            filter.type = 'lowpass';

            filter.frequency.setValueAtTime(800, audioCtx.currentTime);

            filter.frequency.exponentialRampToValueAtTime(80, audioCtx.currentTime + 0.12);

 

            const gain = audioCtx.createGain();

            gain.gain.setValueAtTime(0.7, audioCtx.currentTime);

            gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.12);

 

            noise.connect(filter);

            filter.connect(gain);

            gain.connect(audioCtx.destination);

            noise.start();

        }

 

        function playDing() {

            dingAudio.currentTime = 0;

            dingAudio.play().catch(() => {});

        }

 

        function playIncorrect() {

            incorrectAudio.currentTime = 0;

            incorrectAudio.play().catch(() => {});

        }

 

        // Screen Navigation

        function showScreen(screenId) {

            document.querySelectorAll('.screen').forEach(s => s.classList.remove('active'));

            document.getElementById(screenId).classList.add('active');

 

            if (screenId === 'instruction-screen' || screenId === 'game-screen') {

                if (bgmEnabled) bgmAudio.play().catch(() => {});

            } else {

                bgmAudio.pause();

            }

        }

 

        function shuffle(arr) {

            return arr.sort(() => Math.random() - 0.5);

        }

 

        // Game Logic

        function loadQuestion() {

            if (currentIndex >= transactions.length) {

                endGame();

                return;

            }

 

            currentStage = 'Dr';

            hasFailedCurrent = false;

 

            const item = transactions[currentIndex];

            document.getElementById('question-count').textContent = `Q: ${currentIndex + 1} / ${transactions.length}`;

            document.getElementById('score-display').textContent = `Score: ${score} / ${currentIndex}`;

            document.getElementById('stage-indicator').textContent = 'Dr.';

            document.getElementById('item-display').textContent = item.item;

 

            setupChoices();

        }

 

        function setupChoices() {

            const item = transactions[currentIndex];

            currentChoices = [];

 

            if (currentStage === 'Dr') {

                currentChoices.push(item.dr);

                currentChoices.push(item.cr); // Always include Cr option on Dr entries

               

                let pool = [...item.wrong];

                while (pool.length < 2) {

                    let rand = fallbackAccounts[Math.floor(Math.random() * fallbackAccounts.length)];

                    if (rand !== item.dr && rand !== item.cr && !pool.includes(rand)) pool.push(rand);

                }

                shuffle(pool);

                currentChoices.push(pool[0], pool[1]);

            } else {

                currentChoices.push(item.cr);

               

                let pool = [...item.wrong];

                while (pool.length < 3) {

                    let rand = fallbackAccounts[Math.floor(Math.random() * fallbackAccounts.length)];

                    if (rand !== item.cr && !pool.includes(rand)) pool.push(rand);

                }

                shuffle(pool);

                currentChoices.push(pool[0], pool[1], pool[2]);

            }

 

            shuffle(currentChoices);

 

            document.querySelectorAll('.tangerine-node').forEach((node, i) => {

                node.querySelector('.tangerine-label').textContent = currentChoices[i];

                node.style.display = 'flex';

            });

        }

 

        function handleShot(e) {

            playGunshot();

 

            // Muzzle FX

            const rect = document.getElementById('tree-game-container').getBoundingClientRect();

            const flash = document.createElement('div');

            flash.className = 'muzzle-flash';

            const touch = e.touches ? e.touches[0] : e;

            flash.style.left = `${touch.clientX - rect.left - 22}px`;

            flash.style.top = `${touch.clientY - rect.top - 22}px`;

            document.getElementById('tree-game-container').appendChild(flash);

            setTimeout(() => flash.remove(), 120);

 

            const selected = e.currentTarget.querySelector('.tangerine-label').textContent;

            const item = transactions[currentIndex];

 

            if (currentStage === 'Dr') {

                if (selected === item.dr) {

                    playDing();

                    currentStage = 'Cr';

                    document.getElementById('stage-indicator').textContent = 'Cr.';

                    setupChoices();

                } else {

                    playIncorrect();

                    hasFailedCurrent = true;

                    // Stay on question (Endless trial)

                }

            } else { // Stage === 'Cr'

                if (selected === item.cr) {

                    playDing();

                    if (!hasFailedCurrent) {

                        score++;

                        correctAnswers.push(item);

                    } else {

                        wrongAnswers.push(item);

                    }

                    currentIndex++;

                    setTimeout(loadQuestion, 300);

                } else {

                    playIncorrect();

                    hasFailedCurrent = true;

                    // Stay on question (Endless trial)

                }

            }

        }

 

        function startGame() {

            currentIndex = 0;

            score = 0;

            correctAnswers = [];

            wrongAnswers = [];

            showScreen('game-screen');

            loadQuestion();

        }

 

        function endGame() {

            showScreen('results-screen');

 

            const total = transactions.length;

            const pct = (score / total) * 100;

 

            const endMsg = document.getElementById('end-message');

            if (pct === 100) endMsg.textContent = "Congrats! You are the Dojo Master! 🙇🏻‍♂️";

            else if (pct >= 80) endMsg.textContent = "Closer to becoming the Dojo Master!";

            else if (pct >= 50) endMsg.textContent = "Nice Try!";

            else endMsg.textContent = "Sadly, you need more practice";

 

            document.getElementById('correct-list').innerHTML = correctAnswers.length ?

                correctAnswers.map(a => `<li><b>${a.item}</b><br>Dr. ${a.dr} | Cr. ${a.cr}</li>`).join('') :

                '<li>No first-try correct answers.</li>';

 

            document.getElementById('wrong-list').innerHTML = wrongAnswers.length ?

                wrongAnswers.map(a => `<li><b>${a.item}</b><br>Dr. ${a.dr} | Cr. ${a.cr}</li>`).join('') :

                '<li>Missed 0. Perfect Harvest!</li>';

        }

 

        function copyWrongAnswers() {

            if (wrongAnswers.length === 0) return;

            const text = wrongAnswers.map((a, i) => `${i + 1}. ${a.item}\n   Correct: Dr. ${a.dr} | Cr. ${a.cr}`).join('\n\n');

            navigator.clipboard.writeText(text).then(() => {

                const fb = document.getElementById('copy-feedback');

                fb.style.display = 'block';

                setTimeout(() => fb.style.display = 'none', 800);

            });

        }

 

        // Crosshair movement

        document.addEventListener('mousemove', (e) => {

            const crosshair = document.getElementById('crosshair');

            crosshair.style.display = 'block';

            crosshair.style.left = `${e.clientX}px`;

            crosshair.style.top = `${e.clientY}px`;

        });

 

        // Event Listeners

        document.getElementById('big-tangerine-btn').addEventListener('click', () => showScreen('instruction-screen'));

        document.getElementById('start-harvest-btn').addEventListener('click', startGame);

        document.getElementById('exit-btn').addEventListener('click', endGame);

        document.getElementById('play-again-btn').addEventListener('click', startGame);

        document.getElementById('copy-icon').addEventListener('click', copyWrongAnswers);

 

        document.querySelectorAll('.tangerine-node').forEach(node => {

            node.addEventListener('click', handleShot);

        });
